name: Unzip Project Zip

# This workflow unzips TodoListApp.zip if you uploaded the project as a zip file.
# Put this file at .github/workflows/unzip.yml and push TodoListApp.zip to your repo.
# On the next push it will unzip, commit the files, and build the APK.

on:
  push:
    paths:
      - '**.zip'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  unzip:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          token: ${{ secrets.GITHUB_TOKEN }}
      - name: Unzip if zip exists
        run: |
          ls -la
          if [ -f "TodoListApp.zip" ]; then
            echo "Found TodoListApp.zip - unzipping..."
            unzip -o TodoListApp.zip -d .
            rm TodoListApp.zip
            echo "Unzipped!"
          elif [ -f "BeautifulTodo.zip" ]; then
            unzip -o BeautifulTodo.zip -d .
            rm BeautifulTodo.zip
          else
            echo "No zip found, skipping unzip"
            ls -la
          fi
          ls -R | head -n 100
      - name: Commit unzipped files
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add -A
          git diff --cached --quiet || git commit -m "Unzip TodoListApp project 🤖"
          git push || echo "Nothing to push or push failed - check permissions"
      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'
          cache: gradle
      - name: Build Debug APK (if project exists)
        run: |
          if [ -f "gradlew" ]; then
            chmod +x gradlew
            ./gradlew assembleDebug --stacktrace
          else
            echo "No gradlew found after unzip - check zip structure"
          fi
      - name: Upload APK
        if: success()
        uses: actions/upload-artifact@v4
        with:
          name: BeautifulTodo-debug-apk
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: warn
